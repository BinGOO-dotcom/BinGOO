    <!-- MAIN JAVASCRIPT LOGIC -->
    <script>
        // Game State Variables
        let playerCardData = []; // 5x5 array holding numbers/FREE
        let stampedState = [];   // 5x5 boolean array tracking marked cells
        let hasWonCurrentCard = false;

        // Host Caller State
        let hostRemainingNumbers = [];
        let hostCalledNumbers = new Set();
        let currentCall = null;

        // Column Number Ranges according to standard Bingo rules
        const BINGO_RANGES = {
            'B': { min: 1, max: 15 },
            'I': { min: 16, max: 30 },
            'N': { min: 31, max: 45 },
            'G': { min: 46, max: 60 },
            'O': { min: 61, max: 75 }
        };

        const COL_KEYS = ['B', 'I', 'N', 'G', 'O'];

        // Color maps for stamps per column
        const COL_STAMP_COLORS = [
            'bg-rose-500 border-rose-400 text-white',      // B
            'bg-amber-500 border-amber-400 text-white',    // I
            'bg-emerald-500 border-emerald-400 text-white',// N
            'bg-sky-500 border-sky-400 text-white',        // G
            'bg-purple-500 border-purple-400 text-white'   // O
        ];

        // Initialization on Load
        window.addEventListener('load', () => {
            initPlayerCard();
            initHostMode();
            setupQRCode();
        });

        /* ==========================================================
           1. PLAYER BINGO CARD GENERATION & INTERACTION
           ========================================================== */
        
        // Generates a randomized 5x5 bingo matrix
        function generateRandomCardData() {
            const card = Array.from({ length: 5 }, () => Array(5).fill(null));

            COL_KEYS.forEach((colKey, colIdx) => {
                const range = BINGO_RANGES[colKey];
                const pool = [];
                for (let i = range.min; i <= range.max; i++) pool.push(i);
                
                // Shuffle pool array
                for (let i = pool.length - 1; i > 0; i--) {
                    const j = Math.floor(Math.random() * (i + 1));
                    [pool[i], pool[j]] = [pool[j], pool[i]];
                }

                // Pick first 5 numbers for this column
                for (let rowIdx = 0; rowIdx < 5; rowIdx++) {
                    card[rowIdx][colIdx] = pool[rowIdx];
                }
            });

            // Set Center Cell (Row index 2, Column index 2) to FREE
            card[2][2] = "FREE";
            return card;
        }

        // Initialize or generate a fresh Player Card
        function initPlayerCard() {
            playerCardData = generateRandomCardData();
            
            // Initialize stamp state matrix (Default false, FREE cell true by default)
            stampedState = Array.from({ length: 5 }, () => Array(5).fill(false));
            stampedState[2][2] = true; // FREE space is automatically stamped

            hasWonCurrentCard = false;
            hideWinBanner();
            renderBingoGrid();
        }

        // Render Bingo Grid DOM
        function renderBingoGrid() {
            const gridEl = document.getElementById('bingoGrid');
            gridEl.innerHTML = '';

            for (let r = 0; r < 5; r++) {
                for (let c = 0; c < 5; c++) {
                    const val = playerCardData[r][c];
                    const isStamped = stampedState[r][c];
                    const isFree = (r === 2 && c === 2);

                    // Cell Element
                    const cell = document.createElement('button');
                    cell.type = 'button';
                    cell.onclick = () => toggleStamp(r, c);

                    // Styling based on status
                    let baseStyles = "relative flex flex-col items-center justify-center rounded-2xl font-bold transition-all duration-150 shadow-sm border select-none h-full w-full active:scale-95 ";

                    if (isFree) {
                        baseStyles += isStamped 
                            ? "bg-emerald-600 text-white border-emerald-400 font-extrabold shadow-md " 
                            : "bg-emerald-100 text-emerald-800 border-emerald-300 font-extrabold ";
                    } else if (isStamped) {
                        baseStyles += `${COL_STAMP_COLORS[c]} font-extrabold shadow-md transform scale-[0.98] `;
                    } else {
                        baseStyles += "bg-white/90 hover:bg-white text-slate-800 border-slate-200/80 shadow-slate-200/50 ";
                    }

                    cell.className = baseStyles;

                    // Cell Internal Content
                    if (isFree) {
                        cell.innerHTML = `
                            <span class="text-[10px] sm:text-xs font-black tracking-widest uppercase">FREE</span>
                            <span class="text-xs sm:text-sm">★</span>
                        `;
                    } else {
                        cell.innerHTML = `
                            <span class="text-lg sm:text-2xl font-black leading-none">${val}</span>
                        `;
                    }

                    // Stamp Mark Visual Overlay
                    if (isStamped) {
                        const stampBadge = document.createElement('div');
                        stampBadge.className = "absolute inset-0 m-auto w-8 h-8 sm:w-11 sm:h-11 rounded-full border-2 border-white/60 bg-white/20 flex items-center justify-center stamp-animated pointer-events-none";
                        stampBadge.innerHTML = `<i class="fa-solid fa-check text-white text-xs sm:text-base"></i>`;
                        cell.appendChild(stampBadge);
                    }

                    gridEl.appendChild(cell);
                }
            }
        }

        // Toggle Stamp Mark on cell click
        function toggleStamp(r, c) {
            stampedState[r][c] = !stampedState[r][c];
            renderBingoGrid();
            checkBingoWin();
        }

        // Ask confirmation before clearing card
        function confirmNewCard() {
            if (confirm("Are you sure you want to generate a new card? Current card stamps will be reset.")) {
                initPlayerCard();
                showToast("New Bingo card generated!");
            }
        }

        // Clear player stamps (keep center FREE space marked)
        function clearCardStamps() {
            for (let r = 0; r < 5; r++) {
                for (let c = 0; c < 5; c++) {
                    stampedState[r][c] = (r === 2 && c === 2);
                }
            }
            hasWonCurrentCard = false;
            hideWinBanner();
            renderBingoGrid();
            showToast("Stamps cleared.");
        }

        /* ==========================================================
           2. WIN CHECK & CONFETTI CELEBRATION
           ========================================================== */

        function checkBingoWin() {
            let winLinesCount = 0;

            // Check Rows
            for (let r = 0; r < 5; r++) {
                if (stampedState[r].every(val => val === true)) winLinesCount++;
            }

            // Check Columns
            for (let c = 0; c < 5; c++) {
                let colComplete = true;
                for (let r = 0; r < 5; r++) {
                    if (!stampedState[r][c]) {
                        colComplete = false;
                        break;
                    }
                }
                if (colComplete) winLinesCount++;
            }

            // Check Main Diagonal (Top-Left to Bottom-Right)
            if (stampedState[0][0] && stampedState[1][1] && stampedState[2][2] && stampedState[3][3] && stampedState[4][4]) {
                winLinesCount++;
            }

            // Check Anti-Diagonal (Top-Right to Bottom-Left)
            if (stampedState[0][4] && stampedState[1][3] && stampedState[2][2] && stampedState[3][1] && stampedState[4][0]) {
                winLinesCount++;
            }

            // Trigger celebration on new win
            if (winLinesCount > 0) {
                showWinBanner(`BINGO! (${winLinesCount} Line${winLinesCount > 1 ? 's' : ''})`);
                if (!hasWonCurrentCard) {
                    triggerConfetti();
                    hasWonCurrentCard = true;
                }
            } else {
                hideWinBanner();
                hasWonCurrentCard = false;
            }
        }

        function showWinBanner(text) {
            const banner = document.getElementById('winBanner');
            const winText = document.getElementById('winText');
            winText.textContent = text;
            banner.classList.remove('hidden');
        }

        function hideWinBanner() {
            document.getElementById('winBanner').classList.add('hidden');
        }

        // Fire Confetti Canvas Animation
        function triggerConfetti() {
            if (typeof confetti === 'function') {
                confetti({
                    particleCount: 120,
                    spread: 80,
                    origin: { y: 0.6 }
                });
            }
        }

        /* ==========================================================
           3. HOST / CALLER MODE LOGIC
           ========================================================== */

        function initHostMode() {
            hostRemainingNumbers = [];
            hostCalledNumbers.clear();
            currentCall = null;

            // Populate total 75 numbers with Column Prefix
            COL_KEYS.forEach(key => {
                const range = BINGO_RANGES[key];
                for (let num = range.min; num <= range.max; num++) {
                    hostRemainingNumbers.push({ col: key, number: num });
                }
            });

            // Render Host UI
            updateHostDisplay();
            renderHostHistory();
        }

        function callNextNumber() {
            if (hostRemainingNumbers.length === 0) {
                showToast("All 75 numbers have been called!");
                return;
            }

            // Pick random number from remaining
            const randomIndex = Math.floor(Math.random() * hostRemainingNumbers.length);
            currentCall = hostRemainingNumbers.splice(randomIndex, 1)[0];
            
            // Add to called set
            hostCalledNumbers.add(`${currentCall.col}-${currentCall.number}`);

            // Update Host UI
            updateHostDisplay();
            renderHostHistory();
        }

        function resetHostGame() {
            if (confirm("Are you sure you want to reset the Host Game caller history?")) {
                initHostMode();
                showToast("Host caller game reset!");
            }
        }

        function updateHostDisplay() {
            const displayEl = document.getElementById('currentCallDisplay');
            const counterEl = document.getElementById('callCounter');

            counterEl.textContent = `${hostCalledNumbers.size} / 75`;

            if (currentCall) {
                // Determine color according to column
                let colColorClass = "text-indigo-400";
                if (currentCall.col === 'B') colColorClass = "text-rose-400";
                if (currentCall.col === 'I') colColorClass = "text-amber-400";
                if (currentCall.col === 'N') colColorClass = "text-emerald-400";
                if (currentCall.col === 'G') colColorClass = "text-sky-400";
                if (currentCall.col === 'O') colColorClass = "text-purple-400";

                displayEl.innerHTML = `
                    <span class="${colColorClass} mr-2 font-black">${currentCall.col}</span>
                    <span class="font-black">${currentCall.number}</span>
                `;
            } else {
                displayEl.innerHTML = `<span class="text-slate-600 text-2xl font-semibold">Ready to start?</span>`;
            }
        }

        function renderHostHistory() {
            COL_KEYS.forEach(key => {
                const container = document.getElementById(`hostCol${key}`);
                container.innerHTML = '';

                const range = BINGO_RANGES[key];
                for (let num = range.min; num <= range.max; num++) {
                    const keyStr = `${key}-${num}`;
                    const isCalled = hostCalledNumbers.has(keyStr);

                    const ball = document.createElement('div');
                    ball.className = `py-1 rounded font-bold transition-all text-xs ${
                        isCalled 
                            ? 'bg-indigo-600 text-white shadow-sm font-black' 
                            : 'bg-slate-800/60 text-slate-600 border border-slate-800'
                    }`;
                    ball.textContent = num;
                    container.appendChild(ball);
                }
            });
        }

        /* ==========================================================
           4. SHARE & QR CODE GENERATOR LOGIC
           ========================================================== */

        function setupQRCode() {
            const currentUrl = window.location.href;
            document.getElementById('shareUrlInput').value = currentUrl;

            const qrContainer = document.getElementById('qrcode');
            qrContainer.innerHTML = ''; // Clear existing QR

            try {
                new QRCode(qrContainer, {
                    text: currentUrl,
                    width: 160,
                    height: 160,
                    colorDark : "#0f172a",
                    colorLight : "#ffffff",
                    correctLevel : QRCode.CorrectLevel.H
                });
            } catch (err) {
                // Fallback using simple image QR generator if library has any issues
                qrContainer.innerHTML = `<img src="https://api.qrserver.com/v1/create-qr-code/?size=160x160&data=${encodeURIComponent(currentUrl)}" alt="QR Code" class="w-40 h-40">`;
            }
        }

        function copyShareLink() {
            const shareInput = document.getElementById('shareUrlInput');
            shareInput.select();
            
            try {
                document.execCommand('copy');
                const copyBtnText = document.getElementById('copyBtnText');
                copyBtnText.textContent = "Copied!";
                showToast("Game link copied to clipboard!");
                setTimeout(() => { copyBtnText.textContent = "Copy"; }, 2000);
            } catch (err) {
                showToast("Could not copy link automatically.");
            }
        }

        function openModal(modalId) {
            document.getElementById(modalId).classList.remove('hidden');
        }

        function closeModal(modalId) {
            document.getElementById(modalId).classList.add('hidden');
        }

        // Show brief Toast notification message
        function showToast(message) {
            const toast = document.getElementById('toast');
            const toastMsg = document.getElementById('toastMessage');
            
            toastMsg.textContent = message;
            toast.classList.remove('opacity-0', 'pointer-events-none');
            toast.classList.add('opacity-100');

            setTimeout(() => {
                toast.classList.remove('opacity-100');
                toast.classList.add('opacity-0', 'pointer-events-none');
            }, 2500);
        }
